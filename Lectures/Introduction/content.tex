\begin{frame}[allowframebreaks]{Low-probability but high-cost failures}
In spite of those disturbances, modern electric power systems are very reliable. Assume a duration of power supply interruption of 0.5 hour / year
\begin{center}
\textbf{availability =} (8760$-$0.5)/8760= \textbf{99.9\%} !
\end{center}
However, the cost of unserved energy is high
\begin{itemize}
\item average cost used by CREG (Belgian regulator) to estimate the impact of forced load curtailment: 8.3 k€/MWh (source: Bureau fédéral du plan)
\item varies with time of the day : between 6 and 9 k€/MWh
\item varies with type of consumer : 2.3 k€/MWh for domestic, much higher for industrial
\item even higher average cost considered elsewhere (e.g. 26 k€/MWh in France !)
\end{itemize}
Large-scale failures (blackouts) have tremendous societal consequences
\begin{itemize}
\item next two slides: examples of blackouts and their impacts
\end{itemize}

\section{The power balance problem}

\begin{frame}{The Power Balance Principle}
\begin{columns}
\column{0.44\textwidth}
\begin{figure}
\centering
\includegraphics[width=\linewidth]{images/power_balance.png}
\end{figure}

\column{0.56\textwidth}
Conservation of energy over $\mathrm{d}t$:
\begin{equation*}
  \mathrm{d}E_{\text{gen}}(t) = \mathrm{d}E_{\text{cons}} + \mathrm{d}E_{\text{lost}} + \mathrm{d}E_{\text{net}}
\end{equation*}
Dividing by $\mathrm{d}t$, the instantaneous balance is:
\begin{equation*}
  \mathbf{p_{\text{gen}}(t) = p_{\text{cons}}(t) + p_{\text{lost}}(t) + p_{\text{net}}(t)}
\end{equation*}

\begin{itemize}\footnotesize
\item $p_{\text{cons}}(t)$: Total consumer demand (stochastic, fluctuating).
\item $p_{\text{lost}}(t) = \sum R_k i_k^2$: Joule losses in lines and transformers.
\item $p_{\text{net}}(t)$: Rate of change of electromagnetic energy in $L$ and $C$.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Energy Storage and the AC Network}
\begin{itemize}
\item \textbf{Can we store electrical energy in the AC grid itself?}
  \begin{itemize}
    \item In sinusoidal steady state, power in an inductor or capacitor alternates every quarter cycle ($2\omega$).
    \item In balanced three-phase operation:
    $$\sum_{k=a,b,c} p_{L,k}(t) = 0, \quad \sum_{k=a,b,c} p_{C,k}(t) = 0 \implies \mathbf{p_{\text{net}}(t) \equiv 0}$$
    \item \textbf{Fundamental consequence}: \emph{No net energy can be stored within the AC transmission network!}
  \end{itemize}
\item \textbf{Storage requires conversion to other energy forms}:
  \begin{itemize}
    \item \textbf{Gravitational}: Pumped-storage hydro (e.g. Coo: $1080\text{ MW}$, $>75\%$ efficiency).
    \item \textbf{Electrochemical}: Battery Energy Storage Systems (BESS) (lithium-ion, NaS, flow batteries) expanding exponentially for fast frequency response ($<1\text{ s}$).
    \item \textbf{Chemical}: Electrolysis $\rightarrow$ hydrogen / e-fuels (seasonal storage).
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Need for an Energy Buffer: Power System Inertia}
\begin{itemize}
\item \textbf{The Balancing Challenge}:
  \begin{itemize}
    \item Consumer demand fluctuates continuously.
    \item Primary mechanical energy conversion (steam/gas valves, hydro gates) takes several seconds to react!
  \end{itemize}
\item \textbf{The Rotating Mass Buffer (Inertia)}:
  \begin{itemize}
    \item Massive rotors of synchronous generators provide an instantaneous kinetic energy buffer:
    $$E_{\text{kin}} = \frac{1}{2} J \omega^2$$
    \item Rotor kinetic energy is typically equivalent to $2\text{ to }5\text{ seconds}$ of nominal generator power!
    \item Generation deficit ($p_{\text{gen}} < p_{\text{cons}} + p_{\text{lost}}$) $\rightarrow$ rotors decelerate $\rightarrow$ \textbf{grid frequency drops}.
    \item Generation surplus ($p_{\text{gen}} > p_{\text{cons}} + p_{\text{lost}}$) $\rightarrow$ rotors accelerate $\rightarrow$ \textbf{grid frequency rises}.
  \end{itemize}
\item \textbf{Modern Frontier (Inverter-Based Resources)}:
  \begin{itemize}
    \item Wind, solar, and batteries have no intrinsic physical inertia.
    \item Declining grid inertia requires \textbf{Grid-Forming (GFM)} converter control (covered in Lecture~8 on Frequency Control).
  \end{itemize}
\end{itemize}
\end{frame}

\section{From DC to AC, large interconnections and the come-back of DC}

\begin{frame}
{Network: from early DC ...}
End of 19th century : Gramme, Edison devised the first generators, that produced Direct Current (DC) under relatively low voltages
Impossibility to transmit large powers with direct current:
\begin{itemize}
\item power=voltage×current
\begin{itemize}
\item if the voltage cannot be increased, the current must be
\item but $\text{power lost} = \text{resistance} \times \text{current}^2 \rightarrow$ big waste of energy
\item and large sections of conductors required → expensive and heavy
\end{itemize}
\item Hard to interrupt a large DC current (no zero crossing), for instance after a short-circuit
\end{itemize}
\end{frame}

\begin{frame}
{Network: ... to present high-voltage AC}
Changing for Alternating Current (AC)
\begin{itemize}
\item voltage increased and lowered thanks to the transformer
\item standardized values of frequency : 50 and 60 Hz (other values used at a few places)
\end{itemize}
Larger nominal voltages have been used progressively
\begin{itemize}
\item up to 400 kV in Western Europe
\item up to 765 kV in North America
\item experimental lines at 1100 kV or 1200 kV (Kazakhstan, Japan, etc.)
\end{itemize}
\end{frame}

\begin{frame}
{Motivations for large interconnections}
\begin{itemize}
\item Mutual support between partners to face the loss of generation units
\item Each partner would have to set up a larger “reserve” if it were to operate isolated
\item Larger diversity of energy sources available within the interconnection
\item Allows exploiting complementarity of nuclear, hydro and wind power plants
\item Allows partners to sell/buy energy, to create a large electricity market.
\end{itemize}
\end{frame}

\begin{frame}
{Operational Challenges of Large AC Interconnections}
\begin{itemize}
\item \textbf{Fault Propagation}:
  \begin{itemize}
    \item If one partner is unable to isolate a major contingency, disturbances and cascade trips propagate across the synchronous zone.
  \end{itemize}
\item \textbf{Loop Flows and Parallel Paths ("Wheeling")}:
  \begin{itemize}
    \item Power flows according to Kirchhoff's laws (impedances), not commercial contracts!
    \item Transactions between country A and B induce unscheduled flows through neighboring countries C and D (see next slide).
  \end{itemize}
\item \textbf{Inter-Area Electromechanical Oscillations}:
  \begin{itemize}
    \item Poorly damped low-frequency oscillations ($0.1\text{--}0.5\text{ Hz}$) where rotor angles in one region swing against another region.
  \end{itemize}
\item \textbf{Frequency \& Power Quality Harmonization}:
  \begin{itemize}
    \item Requires strict, shared grid codes and primary frequency reserves across all member TSOs.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Comeback of Direct Current: Modern HVDC}
\begin{itemize}
\item \textbf{Power Electronics Revolution}:
  \begin{itemize}
    \item Transition from thyristors (Line Commutated Converters, LCC) $\rightarrow$ IGBT-based \textbf{Voltage Source Converters (VSC)} and \textbf{Modular Multilevel Converters (MMC)}.
    \item Independent, four-quadrant control of active power $P$ and reactive power $Q$.
    \item Capable of connecting to weak AC grids and providing \textbf{black-start} restoration.
  \end{itemize}
\item \textbf{Key HVDC Transmission Domains}:
  \begin{itemize}
    \item Bulk power transmission over long overhead distances ($>600\text{--}800\text{ km}$).
    \item Subsea and underground high-voltage cables ($>50\text{ km}$).
    \item Controlled asynchronous interconnectors between independent AC synchronous zones.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Transmission through Submarine and Underground Cables}
\begin{columns}
\column{0.55\textwidth}
\begin{itemize}\small
\item \textbf{Why HVDC for cables $>50\text{ km}$?}
  \begin{itemize}\footnotesize
    \item High capacitance of insulated cables produces continuous AC reactive charging current: $I_c = \omega C U$.
    \item Beyond $\approx 50\text{ km}$, charging current exhausts the thermal capacity of the cable!
    \item In DC: $\mathrm{d}U/\mathrm{d}t = 0 \implies$ no steady-state capacitive charging current.
  \end{itemize}
\item \textbf{Operational Interconnectors in Belgium}:
  \begin{itemize}\footnotesize
    \item \textbf{Nemo Link}: $1000\text{ MW}$, $\pm 400\text{ kV}$ subsea link to UK.
    \item \textbf{ALEGrO}: $1000\text{ MW}$, $\pm 320\text{ kV}$ underground link to Germany.
  \end{itemize}
\end{itemize}
