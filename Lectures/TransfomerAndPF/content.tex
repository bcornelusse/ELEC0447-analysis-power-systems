\section{The Transformer}
% Three-phase system slide
\begin{frame}{Basic model of a transformer}
A (single phase) transformer is made of two magnetically coupled \alert{coils} or \alert{windings}.
An ideal transformer is a two-port represented as 
\begin{columns}
    \begin{column}{0.45\textwidth}
        \includegraphics[width=\textwidth]{images/ideal_transformer.png}
    \end{column}
    \begin{column}{0.45\textwidth}
        with
        \begin{align*}
        u_2  &= n u_1 \\
        i_2  &= -\frac{1}{n}i_1
        \end{align*}
        \footnotesize
        \textbf{Sign convention note:} Currents $i_1$ and $i_2$ both enter the positive terminals (standard two-port convention), explaining the minus sign. With load convention ($i_2$ leaving), $i_2 = \frac{1}{n} i_1$.
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Usages}
In power systems, transformers are mainly used to transmit power over long distances by changing the voltage level, 
thus decreasing the current for a given power level. The voltage level of a synchronous generator is around 20kV.

Voltage is changed around five times between generation and load.

It is also used to 
\begin{itemize}
    \item \alert{measure} currents and voltages, 
    \item electrically \alert{isolate} parts of a circuit (not the auto-transformer we will see), 
    \item and \alert{match} impedances.
\end{itemize}
\end{frame}

The ideal model is complemented by elements 
\begin{itemize}
    \item $X_m$ that models the magnetizing inductance
    \item $X_{leakage, i}$ that models the flux not captured by the core on side $i$
    \item $R_{core}$ that models eddy current and hysteresis losses, i.e., losses in the iron core
    \item $R_{1}$ and $R_{2}$ that model (coil) copper losses
\end{itemize}

Parameters are either given in the datasheet or obtained by open-circuit and short-circuit tests.

The core is usually \alert{laminated}  to decrease losses.
\end{frame}

\begin{frame}{Simplification: Neglecting the Excitation Branch}
    Under rated operating conditions, the excitation current $\bar{I}_0$ (magnetizing + core losses) is very small:
    \begin{center}
        $\bar{I}_0 \approx 0.5\%\text{ to }2\% \text{ of rated current}$
    \end{center}
    \begin{columns}
\begin{column}{0.54\textwidth}
    \begin{itemize}
        \item For power flow and short-circuit studies, the shunt branch ($R_{\text{core}} \parallel jX_m$) is \alert{neglected}.
        \item The model reduces to two series winding impedances $Z_p = R_1 + jX_{\text{leakage}, 1}$ and $Z_s = R_2 + jX_{\text{leakage}, 2}$ separated by an ideal transformer ($1:n$).
    \end{itemize}
\end{column}

\end{column}
\end{columns}
\end{frame}

\begin{frame}{Moving Impedances: Referring to the Primary Side}


Refering to the circuit on the previous slide, the voltage seen from primary is: 
$$\bar{V}_p = Z_p \bar{I}_p + \frac{1}{n} \left( Z_s \bar{I}_s + \bar{V}_s\right)
          = \bar{I}_p \left(Z_p + \frac{1}{n^2} Z_s \right) + \frac{\bar{V}_s}{n}$$
since with turns ratio $1:n$,  $\bar{I}_p = n \bar{I}_s \iff \bar{I}_s = \bar{I}_p/n$.

\begin{columns}
\begin{column}{0.56\textwidth}
\begin{itemize}
    \item The total impedance seen from primary is then 
    $$Z_{\text{total}}^{(p)} = Z_p + \frac{Z_s}{n^2}$$
    \item We can remove the impedance $Z_s$ from the secondary and place an impedance $\frac{Z_s}{n^2}$ on the primary side.
\end{itemize}
\normalsize
\end{column}
\begin{column}{0.36\textwidth}

\end{column}
\end{columns}
\end{frame}

\begin{frame}{Physical Intuition: Power Invariance}
\begin{block}{Conservation of Apparent Power Loss}
$$S_{\text{loss}} = I_s^2 Z_s = \left(\frac{I_p}{n}\right)^2 Z_s = I_p^2 \left(\frac{Z_s}{n^2}\right) = I_p^2 Z_s'$$
\end{block}

\vspace{0.2em}
\textbf{Intuition from voltage and current levels:}
\begin{itemize}
    \item In a step-up transformer ($n > 1$), secondary has higher voltage and \alert{lower current}.
    \item For the same power loss or relative drop, a smaller current requires a \alert{larger ohmic impedance} ($Z \propto V^2 / S$).
    \item \textbf{Rule of thumb for referral:}
    \begin{itemize}
        \item Low-voltage $\to$ High-voltage: \textbf{multiply by $n^2$} (impedance increases)
        \item High-voltage $\to$ Low-voltage: \textbf{divide by $n^2$} (impedance decreases)
    \end{itemize}
\end{itemize}
\end{frame}

\section{Per unit representation}

\begin{frame}{Per-Unit representation}
    \begin{block}{Per-unit base selection for transformers}
In power system analysis, base voltages across the transformer are chosen to match the nominal turns ratio:
$$V_{s, \text{base}} = n V_{p, \text{base}} \qquad \text{and} \qquad S_{p, \text{base}} = S_{s, \text{base}} = S_{\text{base}}$$        
    \end{block}


\begin{columns}
\begin{column}{0.4\textwidth}
\textbf{Base Currents:}
$$I_{p, \text{base}} = \frac{S_{\text{base}}}{\sqrt{3}V_{p, \text{base}}} = n I_{s, \text{base}}$$
\small Base currents automatically satisfy the turns ratio!
\end{column} 
\begin{column}{0.52\textwidth}
\textbf{Base Impedances:}
$$Z_{s, \text{base}} = \frac{V_{s, \text{base}}^2}{S_{\text{base}}} = \frac{(n V_{p, \text{base}})^2}{S_{\text{base}}} = \mathbf{n^2 Z_{p, \text{base}}}$$
\small Ratio of impedance bases is \textbf{exactly} $n^2$!
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Per-Unit Representation: Impedance Invariance}
Let us convert the total referred impedance into per unit on each side:

\textbf{1. From primary side (divided by $Z_{p, \text{base}}$):}
$$Z_{\text{tr, pu}}^{(p)} = \frac{Z_{\text{total}}^{(p)}}{Z_{p, \text{base}}} = \frac{Z_p + Z_s/n^2}{Z_{p, \text{base}}} = \frac{Z_p}{Z_{p, \text{base}}} + \frac{Z_s}{n^2 Z_{p, \text{base}}} = \frac{Z_p}{Z_{p, \text{base}}} + \frac{Z_s}{Z_{s, \text{base}}}$$

\textbf{2. From secondary side (divided by $Z_{s, \text{base}}$):}
$$Z_{\text{tr, pu}}^{(s)} = \frac{Z_{\text{total}}^{(s)}}{Z_{s, \text{base}}} = \frac{n^2 Z_p + Z_s}{Z_{s, \text{base}}} = \frac{n^2 Z_p}{n^2 Z_{p, \text{base}}} + \frac{Z_s}{Z_{s, \text{base}}} = \frac{Z_p}{Z_{p, \text{base}}} + \frac{Z_s}{Z_{s, \text{base}}}$$

\vspace{0.3em}
\begin{center}
    \alert{\textbf{The per-unit impedance is strictly identical seen from either side:}}
    $$\mathbf{Z_{\text{tr, pu}} = \frac{Z_{\text{total}}^{(p)}}{Z_{p, \text{base}}} = \frac{Z_{\text{total}}^{(s)}}{Z_{s, \text{base}}} = \frac{Z_p}{Z_{p, \text{base}}} + \frac{Z_s}{Z_{s, \text{base}}}}$$
\end{center}
\end{frame}

\begin{frame}{Per-Unit Representation: Key Consequences}
\begin{columns}[onlytextwidth]
\begin{column}{0.56\textwidth}
\small
\begin{enumerate}
    \item \textbf{Disappearance of Ideal Transformer ($1:1$):}
    $$\bar{V}_{s, \text{pu}} = \bar{V}_{p, \text{pu}}, \qquad \bar{I}_{p, \text{pu}} = \bar{I}_{s, \text{pu}}$$
    The turns ratio becomes $1:1$ and \alert{completely vanishes}.
    \item \textbf{Single Series Branch in $Y_{\text{bus}}$:}
    The transformer is modeled simply as a series branch $Z_{\text{tr, pu}}$ between buses.
    \item \textbf{No Referral Ambiguity:}
    $Z_{\text{tr, pu}}$ is unique (no need to specify "HV" or "LV").
    \item \textbf{Direct Link to Factory Data ($v_k\%$):}
    $v_k\% = 100 \times Z_{\text{tr, pu}}$, no need to separate $Z_p, Z_s$.
\end{enumerate}
\normalsize
\end{column}
\begin{column}{0.42\textwidth}
    \centering
    \input{images/tfo-pu.tex}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{General Per-Unit Base Conversion Formula}
When converting an impedance from $(V_{\text{base}}^{\text{old}}, S_{\text{base}}^{\text{old}})$ to a new base $(V_{\text{base}}^{\text{new}}, S_{\text{base}}^{\text{new}})$:
$$Z_{\text{pu}}^{\text{new}} = Z_{\text{pu}}^{\text{old}} \times \left( \frac{S_{\text{base}}^{\text{new}}}{S_{\text{base}}^{\text{old}}} \right) \times \left( \frac{V_{\text{base}}^{\text{old}}}{V_{\text{base}}^{\text{new}}} \right)^2$$

\begin{itemize}
    \item When voltage bases across transformer zones match the nominal turns ratio:
    $$\frac{V_{\text{base}, 1}}{V_{\text{base}, 2}} = \frac{V_{n1}}{V_{n2}} \implies \frac{V_{\text{base}}^{\text{old}}}{V_{\text{base}}^{\text{new}}} = 1$$
    \item Then $Z_{\text{pu}}$ scales \alert{solely with the MVA base ratio}:
    $$Z_{\text{pu}}^{\text{new}} = Z_{\text{pu}}^{\text{old}} \times \left( \frac{S_{\text{base}}^{\text{new}}}{S_{\text{base}}^{\text{old}}} \right)$$
    This property is what makes per-unit analysis so powerful across multiple voltage tiers.
\end{itemize}
\end{frame}

\begin{itemize}

    \item hence, the total impedance between buses 1 and 2 is $$Z_{12} = 0.0232 + j 0.2608  + 2 * j 0.2pu =  0.0232 + j 0.6608 pu $$

\textbf{In the (345 kV, 100 MVA) base}
\begin{itemize}
    \item the pu value of the impedance is the same in the (\alert{500} kV, 1000 MVA) and (\alert{345} kV, 1000 MVA) bases! 
    \item Why? since we can transfer the impedance from one side of each transformer to the other, cf. a previous remark.
    \item if we now change the MVA base to 100 MVA, 
 $$Z_{12} = (0.0232 + j 0.6608)  \times (100/1000) pu =  0.00232 + j 0.06608 pu$$ since the base impedance is proportional to the inverse of the MVA base.
\end{itemize}
\end{frame}


\section{Transformers' parameters, in practice}
\begin{frame}{Bridging Physics to Datasheets: Why Nameplate Data?}
\footnotesize
The \textbf{nameplate} (\textit{plaque signalétique}) is the certified metal rating plate affixed to the transformer tank, giving guaranteed ratings and factory test values under international standards (IEC 60076 / IEEE C57).

\vspace{0.08cm}
\begin{block}{\small Total Referred Series Impedance is the Universal Link}
\vspace{-0.1cm}
$$Z_k \equiv Z_{\text{total}}^{(p)} = Z_p + \frac{Z_s}{n^2} \qquad \Longleftrightarrow \qquad z_k \equiv Z_{\text{tr, pu}} = \frac{v_k\%}{100}$$
\vspace{-0.2cm}
\centering\scriptsize\textit{(subscript $k$ from German Kurzschluss / French court-circuit $\to$ short-circuit)}
\end{block}

\vspace{0.08cm}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Why not the 6 physical parameters?}
\begin{itemize}\setlength{\itemsep}{2pt}\scriptsize
    \item \textbf{Impractical to separate:} internal parameters ($R_1, R_2, X_{\text{leakage}, 1}, X_{\text{leakage}, 2}$) are sealed in the tank.
    \item \textbf{Aggregated value suffices:} external grid flow and short circuits depend solely on total impedance $Z_k$.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Two Factory Acceptance Tests (IEC 60076):}
\begin{itemize}\setlength{\itemsep}{2pt}\scriptsize
    \item \textbf{Short-Circuit Test:} low voltage, secondary shorted $\to$ core flux negligible $\to$ \alert{isolates series impedance} $Z_k$ ($v_k\%, P_{\text{sc}}$).
    \item \textbf{Open-Circuit Test:} rated voltage, secondary open $\to$ series drop negligible $\to$ \alert{isolates shunt branch} ($P_{\text{fe}}, i_0\%$).
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Short-Circuit Test $\to$ Series Parameters ($r_k, x_k, z_k$)}
\begin{columns}
\begin{column}{0.55\textwidth}
\small
\textbf{Test Setup (IEC 60076):}
\begin{itemize}
    \item Secondary terminals \alert{short-circuited} ($\bar{V}_2 = 0$).
    \item Voltage $V_{\text{sc}}$ applied until rated current $I_{n1}$ flows.
    \item Since $V_{\text{sc}} \approx 8\%\text{--}18\% V_n$, core flux is tiny $\to$ excitation current $\bar{I}_0 \approx 0$.
    \item Applied voltage drops \textbf{entirely across the series windings}:
    $$Z_k = \frac{V_{\text{sc}}}{\sqrt{3} I_{n1}} = \left( R_1 + \frac{R_2}{n^2} \right) + j\left( X_{\text{leakage}, 1} + \frac{X_{\text{leakage}, 2}}{n^2} \right) \equiv Z_{\text{total}}^{(p)}$$
\end{itemize}
\normalsize
\end{column}
\begin{column}{0.45\textwidth}
\small
\begin{block}{Direct Link to Nameplate Data}
\begin{itemize}
    \item \textbf{Impedance voltage $v_k\%$:}
    $$z_{k, \text{tr}} = \frac{Z_k}{Z_n} = \frac{V_{\text{sc}}}{V_n} = \mathbf{\frac{v_k\%}{100}}$$
    \item \textbf{Copper losses $P_{\text{sc}}$ or $v_{kr}\%$:}
    $$r_{k, \text{tr}} = \frac{R_k}{Z_n} = \frac{P_{\text{sc}}}{S_n} = \mathbf{\frac{v_{kr}\%}{100}}$$
    \item \textbf{Total leakage reactance $x_k$:}
    $$x_{k, \text{tr}} = \mathbf{\sqrt{z_{k, \text{tr}}^2 - r_{k, \text{tr}}^2}}$$
\end{itemize}
\end{block}
\normalsize
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Open-Circuit Test $\to$ Shunt Parameters ($R_{\text{core}}, X_m$)}
\begin{columns}
\begin{column}{0.54\textwidth}
\small
\textbf{Test Setup (IEC 60076):}
\begin{itemize}
    \item Secondary terminals \alert{open-circuited} ($I_2 = 0$).
    \item Rated voltage $V_n$ applied to primary.
    \item Current is only no-load excitation current $\bar{I}_0$ ($< 0.5\%$ of $I_n$).
    \item Series drops ($I_0 R_1, I_0 X_{\text{leakage}, 1}$) are negligible $\to$ rated voltage falls \textbf{entirely across the shunt branch} ($R_{\text{core}} \parallel jX_m$).
\end{itemize}
\normalsize
\end{column}
\begin{column}{0.46\textwidth}
\small
\begin{block}{Link to Nameplate Data}
\begin{itemize}
    \item \textbf{Iron / core losses $P_{\text{fe}}$:}
    $$P_{\text{fe}} = \frac{V_n^2}{R_{\text{core}}} \implies g_{\text{fe, tr}} = \mathbf{\frac{P_{\text{fe}}}{S_n} \text{ [pu]}}$$
    \item \textbf{No-load current $i_0\%$:}
    $$I_0 \approx \frac{V_n/\sqrt{3}}{X_m} \implies b_{m, \text{tr}} \approx \mathbf{\frac{i_0\%}{100} \text{ [pu]}}$$
\end{itemize}
\end{block}
\vspace{-0.2em}
\begin{alertblock}{Neglected in Power Flow}
\footnotesize
Since $i_0\% \le 0.5\% \implies x_m \ge 200\text{ pu}$, shunt branch is usually omitted in transmission grid studies.
\end{alertblock}
\normalsize
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Efficiency}

The efficiency expressed in \% is 
$$100 \times \frac{P_{output}}{P_{input}} = 100 \times \left(1 - \frac{P_{losses}}{P_{input}}\right) $$

Remarks:

    \item maximal when loaded such that copper losses = iron losses (cancel derivative of efficiency w.r.t current) 
    \item Around \alert{99.5 \%} in large power transformers at full load.
\end{itemize}
\end{frame}

\begin{frame}{Auto-transformers: Sizing \& Why $V_H/V_L \approx 1$ Matters}
\small
The \textbf{core sizing factor} (co-ratio) defines the fraction of power crossing the core:
$$k_{\text{size}} = \frac{S_{\text{ind}}}{S_{\text{rated}}} = 1 - \frac{V_L}{V_H} = \frac{V_H - V_L}{V_H} \qquad (k_{\text{size}} = 1 \text{ for 2-winding transformer})$$

\vspace{-0.1cm}
\begin{columns}[T]
    \begin{column}{0.50\textwidth}
        \textbf{When $V_H/V_L \approx 1$ $\implies$ High Savings ($k_{\text{size}} \ll 1$):}
        \begin{itemize}\setlength{\itemsep}{2pt}\footnotesize
            \item \textbf{Grid tie ($380 / 220\text{ kV}$, $V_H/V_L \approx 1.73$):}\\
            $k_{\text{size}} = 1 - 220/380 = \mathbf{0.42}$\\
            $\rightarrow$ \textbf{1000 MVA} throughput needs only a \textbf{420 MVA} core! ($58\%$ saved in core size, weight, losses).
            \item \textbf{OLTC Regulation ($\pm 10\%$ boost, $V_H/V_L = 1.1$):}\\
            $k_{\text{size}} = 1 - 1/1.10 \approx \mathbf{0.09} \implies$ core handles only \textbf{9\%}!
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{When $V_H/V_L \gg 1$ $\implies$ Negligible Savings:}
        \begin{itemize}\setlength{\itemsep}{2pt}\footnotesize
            \item \textbf{Generator Step-Up ($20 / 400\text{ kV}$):}\\
            $k_{\text{size}} = 1 - 20/400 = \mathbf{0.95}$ (only 5\% savings).
            \item Losing galvanic isolation is not worth a $5\%$ saving $\rightarrow$ standard 2-winding used instead!
        \end{itemize}
        
        \vspace{0.15cm}
        \alert{\textbf{Rule of thumb:}} Auto-transformers are economically justified when $\mathbf{V_H / V_L \le 2}$.
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Auto-transformers: Trade-offs \& Applications}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \textbf{Advantages:}
        \begin{itemize}
            \item Physically smaller and lighter
            \item Less costly (significantly less copper \& iron)
            \item Higher efficiency ($\eta > 99.7\%$)
            \item Lower leakage reactance
            \item Easy to implement tap changers (OLTC)
            \item Common solid earth ground
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{Disadvantages:}
        \begin{itemize}
            \item No galvanic / electrical isolation
            \item Higher short-circuit current
            \item Risk of full $V_H$ on secondary if common winding opens
            \item Economic benefit diminishes if ratio $V_H/V_L > 2$
        \end{itemize}
    \end{column}
\end{columns}

\vspace{0.25cm}
\footnotesize
\textbf{Typical Applications:} Interconnecting close transmission voltage levels (e.g. 380/220 kV, 220/150 kV) and serving as the regulating unit in on-load tap changers and phase shifters.
\end{frame}

\begin{frame}{Phase shift in delta-star transformers}

The star part has $n$ times the number of turns of the delta part (primary side).

    \item Since  $\bar{V}_{a,s} = n \bar{V}_{AC,p}$, $\bar{V}_{a,s} = n\sqrt{3} \bar{V}_{a,p} \angle{-30^\circ}$
\end{itemize}

\begin{frame}{Standard Three-Phase Vector Groups (IEC 60076)}
In industrial practice, phase displacement between windings is designated by a \alert{clock number} (each hour represents a $30^\circ$ lag):

\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \small
        \textbf{Notation Rules:}
        \begin{itemize}
            \item 1st letter: HV winding (\textbf{D} = Delta, \textbf{Y} = Wye)
            \item 2nd letter: LV winding (\textbf{d} = delta, \textbf{y} = wye, \textbf{n} = neutral out)
            \item Number $k \in \{0, \dots, 11\}$: phase lag of LV line-to-neutral phasor relative to HV:
            $\theta_{\text{LV}} - \theta_{\text{HV}} = - k \times 30^\circ$
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \small
        \textbf{Common Examples:}
        \begin{itemize}
            \item \textbf{Yy0 / Dd0}: $0^\circ$ shift (12 o'clock)
            \item \textbf{Dy1}: Delta-Wye, LV lags by $30^\circ$ (1 o'clock) --- \textit{the case studied on previous slide!}
            \item \textbf{Dy11} (or \textbf{Yd11}): LV leads by $30^\circ$ / lags by $330^\circ$ (11 o'clock)
            \item \textbf{Yd5}: LV lags by $150^\circ$ (5 o'clock)
        \end{itemize}
    \end{column}
\end{columns}

\vspace{0.15cm}
\footnotesize
\textbf{Paralleling Condition:} Transformers can be operated in parallel only if they belong to the \alert{same vector group} (identical phase displacement).
\end{frame}

\begin{frame}[allowframebreaks]{Power flow regulation by phase shifting}

We have seen that active power flows are dictated by the voltage magnitudes but also the sine of the angle difference between buses: 

\begin{columns}
    \begin{column}{0.55\textwidth}
        \begin{center}
            \includegraphics[width=\textwidth]{images/power-transfer_AC.png}\\
        \end{center}
    \end{column}
    \begin{column}{0.45\textwidth}
        \begin{align*}
        S_r &= \bar{V}_r\bar{I}^*  = V_r \left(\frac{V_s \angle -\theta - V_r}{-jX}\right) \\\\
            &= \frac{V_s V_r \sin \theta }{X} +j \frac{V_s V_r \cos \theta - V^2_r}{X} 
        \end{align*}
        $\theta$ is the angle between $\bar{V}_r$ and $\bar{V}_s$
    \end{column}
\end{columns}

In per unit ($V_s \approx V_r \approx 1$), if the angle difference is small (DC approximation), we have $$P_r = \frac{\theta_r - \theta_s}{X}.$$

If we have a device that can generate an adjustable phase shift, we can control the power flows.
This is the purpose of \alert{phase-shifting transformers}.
\end{frame}

Active power divides inversely proportional to line reactances: $P \sim \Delta \theta / X$.
In meshed grids, this creates \alert{transit flows} and line overloads.

\section{Transformers in the power flow analysis}

\begin{frame}{Transformer without regulation}


A transformer, in the per-unit representation, can thus be represented 
\begin{itemize}
    \item as a two-port if the shunt admittance is considered
    \item as a simple series leakage impedance if the shunt admittance is neglected
\end{itemize}
 
It is a bit harder to model controllable taps for voltage and flow control.

\end{frame}

\begin{frame}[allowframebreaks]{Representing taps and phase shifts}

\begin{columns}
    \begin{column}{0.5\textwidth}
    Let $Y_l$ be the leakage admittance and $t$ be the off-nominal turns ratio:
    \begin{itemize}
        \item If $t \in \mathbb{R}$ (typically $0.90 \le t \le 1.10$, step size $\approx 0.5\%\text{--}1.25\%$), this corresponds to an \alert{in-phase tap-changer (OLTC)} for voltage magnitude and reactive power control.
        \item If $t = m e^{j\theta}$ with $\theta \ne 0$ ($|\theta| \le 15^\circ\text{--}25^\circ$), this is a \alert{phase-shifting transformer (PST)} for active power flow control.
    \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
        \includegraphics[width=\textwidth]{images/tap-shift-tfo.png}
    \end{column}
\end{columns}

We have 
$$\bar{I}_1 = \left(\bar{V}_1 - \frac{\bar{V}_2}{t}\right) Y_l$$
and since $\frac{\bar{V}_2}{t} \bar{I}_1^\star = -\bar{V}_2 \bar{I}_2^\star$ by energy conservation
$$\bar{I}_2 = - \frac{\bar{I}_1}{t^\star} = - \bar{V}_1 \frac{Y_l}{t^\star} + \bar{V}_2 \frac{Y_l}{|t|^2}$$

Thus tap and phase shift can be represented by the two-port admittance matrix:
$$\begin{bmatrix} \bar{I}_1 \\ \bar{I}_2\end{bmatrix} 
 = \begin{bmatrix} Y_l & -\dfrac{Y_l}{t} \\ -\dfrac{Y_l}{t^\star} & \dfrac{Y_l}{|t|^2} \end{bmatrix}
 \begin{bmatrix} \bar{V}_1 \\ \bar{V}_2\end{bmatrix}$$
\end{frame}

\begin{frame}[allowframebreaks]{System $\mathbf{Y}_{\text{bus}}$ Stamping \& Newton-Raphson Impact}
For a transformer connected between \alert{Bus $i$} (series impedance side) and \alert{Bus $j$} (ideal tap side $1:t$):

$$\begin{bmatrix} \bar{I}_i \\ \bar{I}_j \end{bmatrix} = 
\begin{bmatrix} 
Y_l & -\dfrac{Y_l}{t} \\ 
-\dfrac{Y_l}{t^\star} & \dfrac{Y_l}{|t|^2} 
\end{bmatrix}
\begin{bmatrix} \bar{V}_i \\ \bar{V}_j \end{bmatrix}$$

\textbf{Stamping Rule into the Global $\mathbf{Y}_{\text{bus}}$ Matrix:}
\begin{align*}
    Y_{ii} &\leftarrow Y_{ii} + Y_l \\
    Y_{jj} &\leftarrow Y_{jj} + \frac{Y_l}{|t|^2} \\
    Y_{ij} &\leftarrow Y_{ij} - \frac{Y_l}{t} \\
    Y_{ji} &\leftarrow Y_{ji} - \frac{Y_l}{t^\star}
\end{align*}

\newpage
\begin{alertblock}{Consequence for Complex Taps ($t = m e^{j\theta}, \theta \ne 0$)}
Because $t \ne t^\star$, the off-diagonal terms differ:
$$Y_{ij} \ne Y_{ji} \implies \mathbf{Y}_{\text{bus}} \ne \mathbf{Y}_{\text{bus}}^T \quad \text{(\alert{loss of symmetry!})}$$
\end{alertblock}

\textbf{Direct Impact on the Newton-Raphson Power Flow:}
\begin{itemize}
    \item Standard textbook formulations often assume $Y_{km} = Y_{mk}$ ($G_{km} = G_{mk}, B_{km} = B_{mk}$).
    \item With PSTs, one \alert{must} strictly use directed branch admittance values:
    $$P_k = \sum_{m=1}^n V_k V_m \left( G_{km} \cos(\theta_k - \theta_m) + B_{km} \sin(\theta_k - \theta_m) \right)$$
    where $G_{km} + j B_{km} = Y_{km} \ne Y_{mk}$.
    \item Off-diagonal blocks in the Jacobian $\mathbf{J}$ are no longer transpose-symmetric.
\end{itemize}
\end{frame}
