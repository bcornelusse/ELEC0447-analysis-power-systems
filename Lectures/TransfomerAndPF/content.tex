\section{The Transformer}
% Three-phase system slide
\begin{frame}{Basic model of a transformer}
A (single phase) transformer is made of two magnetically coupled \alert{coils} or \alert{windings}.
An ideal transformer is a two-port represented as 
\begin{columns}
    \begin{column}{0.45\textwidth}
        \includegraphics[width=\textwidth]{images/ideal_transformer.png}
    \end{column}
    \begin{column}{0.45\textwidth}
        with
        \begin{align*}
        u_2  &= n u_1 \\
        i_2  &= -\frac{1}{n}i_1
        \end{align*}
        \footnotesize
        \textbf{Sign convention note:} Currents $i_1$ and $i_2$ both enter the positive terminals (standard two-port convention), explaining the minus sign. With load convention ($i_2$ leaving), $i_2 = \frac{1}{n} i_1$.
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Usages}
In power systems, transformers are mainly used to transmit power over long distances by changing the voltage level, 
thus decreasing the current for a given power level. The voltage level of a synchronous generator is around 20kV.

Voltage is changed around five times between generation and load.

It is also used to 
\begin{itemize}
    \item \alert{measure} currents and voltages, 
    \item electrically \alert{isolate} parts of a circuit (not the auto-transformer we will see), 
    \item and \alert{match} impedances.
\end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Non-ideal model}

\begin{center}
   \includegraphics[width=0.9\textwidth]{images/non-ideal-transformer.png}
\end{center}
The ideal model is complemented by elements 
\begin{itemize}
    \item $X_m$ that models the magnetizing inductance
    \item $X_{leakage, i}$ that models the flux not captured by the core on side $i$
    \item $R_{core}$ that models eddy current and hysteresis losses, i.e., losses in the iron core
    \item $R_{1}$ and $R_{2}$ that model (coil) copper losses
\end{itemize}

Parameters are either given in the datasheet or obtained by open-circuit and short-circuit tests.

The core is usually \alert{laminated}  to decrease losses.
\end{frame}

\begin{frame}{Simplification}

\begin{columns}
\begin{column}{0.45\textwidth}
    The excitation current, the sum of the currents in $R_{core}$ and $X_m$, is often neglected, 
    leading to a simpler non-ideal model, and the series impedances can be transferred from one side to the other, 
    with $$Z_p = R_1 + jX_{leakage, 1}$$ and $$Z_s = R_2 + jX_{leakage, 2}$$
\end{column}
\begin{column}{0.5\textwidth}
    \includegraphics[width=0.8\textwidth]{images/non-ideal-transformer-2.png}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[allowframebreaks]{Per unit representation}
Let's consider the rated voltages and currents on both sides of the (ideal) transformer as base values. As
$$V_{s, base} = n V_{p, base} $$ and $$ I_{p, base} = n I_{s, base},$$
the \alert{MVA base is the same on both sides}, and thus
$$Z_{s, base} = n^2 Z_{p, base}$$


\begin{columns}
    \begin{column}{0.35\textwidth}
        Hence, \alert{in per unit, the transformer can be replaced by a single impedance}
        $$Z_{tr} = \frac{Z_{p}}{Z_{p, base}}+\frac{Z_{s}}{Z_{s, base}}.$$
    \end{column}
    \begin{column}{0.6\textwidth}
        \includegraphics[width=\textwidth]{images/tfo-pu.png}
    \end{column}
\end{columns}

Thus we have also that 
\begin{align*}
    Z_{tr} &= \frac{Z_{p}+ Z_s/n^2}{Z_{p, base}} \\
     &= \frac{n^2 Z_{p} + Z_s}{Z_{s, base}}
\end{align*}
i.e. \alert{the impedance is the same whether we see it from the primary or the secondary side, although the voltage bases differ}.

Also, if the three-phase transformer is \alert{wye-delta} connected, a \alert{30° phase shift} must be applied (more on this later).
\end{frame}

\begin{frame}[allowframebreaks]{From Nameplate Data to Per-Unit Parameters}
In manufacturer datasheets and power flow software (e.g., PandaPower), transformers are specified via standard factory test values:
\begin{itemize}
    \item \textbf{Ratings:} rated power $S_n$ [MVA] and rated voltages $V_{n1}, V_{n2}$ [kV]
    \item \textbf{Short-circuit test (copper losses \& leakage):}
    \begin{itemize}
        \item Short-circuit voltage: $v_k\%$ (typically $8\%\text{ to }18\%$ for transmission units)
        \item Ohmic drop / copper losses: $v_{kr}\%$ (or short-circuit active power $P_{\text{sc}}$ in kW)
    \end{itemize}
    \item \textbf{Open-circuit / no-load test (iron losses \& magnetizing):}
    \begin{itemize}
        \item No-load active losses: $P_{\text{fe}}$ [kW] (or $p_0\%$)
        \item No-load magnetizing current: $i_0\%$ (typically $< 0.5\%$, usually neglected in power flow)
    \end{itemize}
\end{itemize}

\newpage
\textbf{1. On the transformer's own rated base $(V_n, S_n)$:}
$$z_{k, \text{tr}} = \frac{v_k\%}{100} \text{ pu}, \qquad r_{k, \text{tr}} = \frac{v_{kr}\%}{100} = \frac{P_{\text{sc}}}{S_n} \text{ pu}, \qquad x_{k, \text{tr}} = \sqrt{z_{k, \text{tr}}^2 - r_{k, \text{tr}}^2} \text{ pu}$$

\textbf{2. Scaling to the system network base $(V_{\text{base}}, S_{\text{base}})$:}
$$Z_{k, \text{sys}} = \left( r_{k, \text{tr}} + j x_{k, \text{tr}} \right) \times \left( \frac{S_{\text{base}}}{S_n} \right) \times \left( \frac{V_n}{V_{\text{base}}} \right)^2$$
Since $X_{k} \gg R_{k}$ in large power transformers (often $X/R > 20$), $R_k$ is often neglected in simplified transmission calculations ($Z_{\text{tr}} \approx j x_k$).

\begin{block}{PandaPower Implementation}
\small
In PandaPower, this conversion is computed directly from parameters:
\begin{align*}
\texttt{z\_k} &= (\texttt{vk\_percent} / 100) \times (\texttt{net.sn\_mva} / \texttt{sn\_mva}) \\
\texttt{r\_k} &= (\texttt{vkr\_percent} / 100) \times (\texttt{net.sn\_mva} / \texttt{sn\_mva}) \\
\texttt{x\_k} &= \sqrt{\texttt{z\_k}^2 - \texttt{r\_k}^2}
\end{align*}
\end{block}
\end{frame}

\begin{frame}{General Per-Unit Base Conversion Formula}
When converting an impedance from $(V_{\text{base}}^{\text{old}}, S_{\text{base}}^{\text{old}})$ to a new base $(V_{\text{base}}^{\text{new}}, S_{\text{base}}^{\text{new}})$:
$$Z_{\text{pu}}^{\text{new}} = Z_{\text{pu}}^{\text{old}} \times \left( \frac{S_{\text{base}}^{\text{new}}}{S_{\text{base}}^{\text{old}}} \right) \times \left( \frac{V_{\text{base}}^{\text{old}}}{V_{\text{base}}^{\text{new}}} \right)^2$$

\begin{itemize}
    \item When voltage bases across transformer zones match the nominal turns ratio:
    $$\frac{V_{\text{base}, 1}}{V_{\text{base}, 2}} = \frac{V_{n1}}{V_{n2}} \implies \frac{V_{\text{base}}^{\text{old}}}{V_{\text{base}}^{\text{new}}} = 1$$
    \item Then $Z_{\text{pu}}$ scales \alert{solely with the MVA base ratio}:
    $$Z_{\text{pu}}^{\text{new}} = Z_{\text{pu}}^{\text{old}} \times \left( \frac{S_{\text{base}}^{\text{new}}}{S_{\text{base}}^{\text{old}}} \right)$$
    This property is what makes per-unit analysis so powerful across multiple voltage tiers.
\end{itemize}
\end{frame}

\begin{itemize}
    \item $Z_{line, pu} = 200 \times (0.029 + j 0.326) / (500^2/1000) = 0.0232 + j 0.2608 pu$
    \item hence, the total impedance between buses 1 and 2 is $$Z_{12} = 0.0232 + j 0.2608  + 2 * j 0.2pu =  0.0232 + j 0.6608 pu $$
\end{itemize}


\textbf{In the (345 kV, 100 MVA) base}
\begin{itemize}
    \item the pu value of the impedance is the same in the (\alert{500} kV, 1000 MVA) and (\alert{345} kV, 1000 MVA) bases! 
    \item Why? since we can transfer the impedance from one side of each transformer to the other, cf. a previous remark.
    \item if we now change the MVA base to 100 MVA, 
 $$Z_{12} = (0.0232 + j 0.6608)  \times (100/1000) pu =  0.00232 + j 0.06608 pu$$ since the base impedance is proportional to the inverse of the MVA base.
\end{itemize}

\begin{frame}{Efficiency}

The efficiency expressed in \% is 
$$100 \times \frac{P_{output}}{P_{input}} = 100 \times \left(1 - \frac{P_{losses}}{P_{input}}\right) $$

Remarks: 
\begin{itemize}
    \item maximal when loaded such that copper losses = iron losses (cancel derivative of efficiency w.r.t current) 
    \item Around \alert{99.5 \%} in large power transformers at full load.
\end{itemize}
\end{frame}

\begin{frame}{Auto-transformers: Trade-offs \& Applications}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \textbf{Advantages:}
        \begin{itemize}
            \item Physically smaller and lighter
            \item Less costly (significantly less copper \& iron)
            \item Higher efficiency ($\eta > 99.7\%$)
            \item Lower leakage reactance
            \item Easy to implement tap changers (OLTC)
            \item Common solid earth ground
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{Disadvantages:}
        \begin{itemize}
            \item No galvanic / electrical isolation
            \item Higher short-circuit current
            \item Risk of full $V_H$ on secondary if common winding opens
            \item Economic benefit diminishes if ratio $V_H/V_L > 2$
        \end{itemize}
    \end{column}
\end{columns}

\vspace{0.25cm}
\footnotesize
\textbf{Typical Applications:} Interconnecting close transmission voltage levels (e.g. 380/220 kV, 220/150 kV) and serving as the regulating unit in on-load tap changers and phase shifters.
\end{frame}

\begin{frame}{Phase shift in delta-star transformers}

The star part has $n$ times the number of turns of the delta part (primary side).


\begin{columns}
    \begin{column}{0.45\textwidth}
Let's reason on phase $a$,
\begin{itemize}
    \item Voltage $\bar{V}_{a,s}$ is on the same core as $\bar{V}_{AC,p} = \sqrt{3}\bar{V}_{a,p} \angle{-30^\circ}$ where $\bar{V}_{a,p}$ is the (virtual) phase-neutral voltage on the primary side.
    \item Since  $\bar{V}_{a,s} = n \bar{V}_{AC,p}$, $\bar{V}_{a,s} = n\sqrt{3} \bar{V}_{a,p} \angle{-30^\circ}$
\end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
\begin{center}
    \includegraphics[width=\textwidth]{images/tfo-delta-star.png}\\
\end{center}
    \end{column}
\end{columns}

We \alert{gain a $\sqrt{3}$ factor} in the amplification, and a \alert{lagging phase shift of 30°}.

The same reasoning holds for phases $b$ and $c$.

\end{frame}

\begin{frame}{Standard Three-Phase Vector Groups (IEC 60076)}
In industrial practice, phase displacement between windings is designated by a \alert{clock number} (each hour represents a $30^\circ$ lag):

\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \small
        \textbf{Notation Rules:}
        \begin{itemize}
            \item 1st letter: HV winding (\textbf{D} = Delta, \textbf{Y} = Wye)
            \item 2nd letter: LV winding (\textbf{d} = delta, \textbf{y} = wye, \textbf{n} = neutral out)
            \item Number $k \in \{0, \dots, 11\}$: phase lag of LV line-to-neutral phasor relative to HV:
            $\theta_{\text{LV}} - \theta_{\text{HV}} = - k \times 30^\circ$
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \small
        \textbf{Common Examples:}
        \begin{itemize}
            \item \textbf{Yy0 / Dd0}: $0^\circ$ shift (12 o'clock)
            \item \textbf{Dy1}: Delta-Wye, LV lags by $30^\circ$ (1 o'clock) --- \textit{the case studied on previous slide!}
            \item \textbf{Dy11} (or \textbf{Yd11}): LV leads by $30^\circ$ / lags by $330^\circ$ (11 o'clock)
            \item \textbf{Yd5}: LV lags by $150^\circ$ (5 o'clock)
        \end{itemize}
    \end{column}
\end{columns}

\vspace{0.15cm}
\footnotesize
\textbf{Paralleling Condition:} Transformers can be operated in parallel only if they belong to the \alert{same vector group} (identical phase displacement).
\end{frame}

\begin{frame}[allowframebreaks]{Power flow regulation by phase shifting}

We have seen that active power flows are dictated by the voltage magnitudes but also the sine of the angle difference between buses: 

\begin{columns}
    \begin{column}{0.55\textwidth}
        \begin{center}
            \includegraphics[width=\textwidth]{images/power-transfer_AC.png}\\
        \end{center}
    \end{column}
    \begin{column}{0.45\textwidth}
        \begin{align*}
        S_r &= \bar{V}_r\bar{I}^*  = V_r \left(\frac{V_s \angle -\delta - V_r}{-jX}\right) \\\\
            &= \frac{V_s V_r \sin \delta }{X} +j \frac{V_s V_r \cos \delta - V^2_r}{X} 
        \end{align*}
        $\delta$ is the angle between $\bar{V}_r$ and $\bar{V}_s$
    \end{column}
\end{columns}

If we have a device that can generate an adjustable phase shift, we can control the power flows.
This is the purpose of \alert{phase-shifting transformers}.

In practice phase shifting is achieved by "combining the signal with a fraction of itself shifted by 90°".
For the details of how it is implemented or modeled, see 
\begin{itemize}
    \item \href{https://en.wikipedia.org/wiki/Quadrature_booster}{\underline{Wikipedia}}
    \item Section 5.7 of Weedy or ELEC0014.
    \item \href{https://eepublicdownloads.blob.core.windows.net/public-cdn-container/clean-documents/CIM_documents/Grid_Model_CIM/ENTSOE_CGMES_v2.4.14_28May2014_PSTmodelling.pdf}{\underline{ENTSO-E - Phase Shift Transformers Modelling, Version 1.0.0, May 2015}}
\end{itemize}

\end{frame}

\begin{frame}{Physical Principle: Controlling Parallel Path \& Loop Flows}
Active power divides inversely proportional to line reactances: $P \sim \Delta \theta / X$.
In meshed grids, this creates \alert{transit flows} and line overloads.

\begin{columns}[T]
    \begin{column}{0.54\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.72, >=Stealth]
            % Buses
            \draw[line width=2.5pt] (0, -1.0) -- (0, 1.0) node[above, font=\bfseries] {Area A};
            \draw[line width=2.5pt] (5.2, -1.0) -- (5.2, 1.0) node[above, font=\bfseries] {Area B};
            
            % Path 1 (Direct, low impedance)
            \draw[line width=1pt] (0, 0.6) -- (1.5, 0.6) node[midway, above, font=\footnotesize] {$X_1$ (small)};
            \draw[fill=orange!25, line width=1pt] (1.5, 0.25) rectangle (3.2, 0.95) node[midway, font=\scriptsize\bfseries] {PST ($\pm\alpha$)};
            \draw[line width=1pt, ->] (3.2, 0.6) -- (5.2, 0.6) node[midway, above, font=\footnotesize] {$P_1(\alpha)$};
            
            % Path 2 (Alternative path)
            \draw[line width=1pt, ->] (0, -0.6) -- (5.2, -0.6) node[midway, below, font=\footnotesize] {Path 2: $X_2$ (higher reactance)};
            
            % Injections
            \draw[line width=1.2pt, ->] (-0.9, 0) -- (0, 0) node[midway, above, font=\footnotesize] {$P_{\text{total}}$};
            \draw[line width=1.2pt, ->] (5.2, 0) -- (6.1, 0) node[midway, above, font=\footnotesize] {$P_{\text{total}}$};
        \end{tikzpicture}
        \end{center}
    \end{column}
    \begin{column}{0.46\textwidth}
        \small
        \textbf{Without PST ($\alpha = 0$):}
        $$P_1 = P_{\text{total}} \frac{X_2}{X_1 + X_2} \quad (\text{overload on Line 1!})$$
        
        \textbf{With PST ($\alpha \ne 0$):}
        $$P_1 \approx \frac{\theta_A - \theta_B + \alpha}{X_1}$$
        Tuning $\alpha$ redirects power into Path 2, relieving Line 1 without costly generation redispatch!
    \end{column}
\end{columns}

\vspace{0.1cm}
\footnotesize
\textbf{Why PSTs on Belgian borders?} German wind power flowing to Southern Germany creates large transit loop flows across Belgium/Netherlands. PSTs act as electrical valves to protect interconnectors.
\end{frame}

\begin{frame}{Remarks}

In \alert{three-phase operation},
\begin{itemize}
\item either there are three separate single-phase transformers (easier to fix when there is a problem on a phase, more modular)
\item or a \alert{three-phase transformer}, that is a single core with three auto-transformers on it, cf. the video at the beginning of this presentation (cheaper, lighter core and less copper).
\end{itemize}
 
Some transformers called \alert{three-winding transformers} are equipped with a third winding (not to be confused with a three-phase transformer) that is used for auxiliary purposes (feeding auxiliary devices e.g., fans, providing reactive power support, ...).


\end{frame}

\section{Transformers in the power flow analysis}

\begin{frame}{Transformer without regulation}


A transformer, in the per-unit representation, can thus be represented 
\begin{itemize}
    \item as a two-port if the shunt admittance is considered
    \item as a simple series leakage impedance if the shunt admittance is neglected
\end{itemize}
 
It is a bit harder to model controllable taps for voltage and flow control.

\end{frame}

\begin{frame}[allowframebreaks]{Representing taps and phase shifts}

\begin{columns}
    \begin{column}{0.5\textwidth}
    Let $Y_l$ be the leakage admittance and $t$ be the off-nominal turns ratio:
    \begin{itemize}
        \item If $t \in \mathbb{R}$ (typically $0.90 \le t \le 1.10$, step size $\approx 0.5\%\text{--}1.25\%$), this corresponds to an \alert{in-phase tap-changer (OLTC)} for voltage magnitude and reactive power control.
        \item If $t = m e^{j\theta}$ with $\theta \ne 0$ ($|\theta| \le 15^\circ\text{--}25^\circ$), this is a \alert{phase-shifting transformer (PST)} for active power flow control.
    \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
        \includegraphics[width=\textwidth]{images/tap-shift-tfo.png}
    \end{column}
\end{columns}

We have 
$$\bar{I}_1 = \left(\bar{V}_1 - \frac{\bar{V}_2}{t}\right) Y_l$$
and since $\frac{\bar{V}_2}{t} \bar{I}_1^\star = -\bar{V}_2 \bar{I}_2^\star$ by energy conservation
$$\bar{I}_2 = - \frac{\bar{I}_1}{t^\star} = - \bar{V}_1 \frac{Y_l}{t^\star} + \bar{V}_2 \frac{Y_l}{|t|^2}$$

Thus tap and phase shift can be represented by the two-port admittance matrix:
$$\begin{bmatrix} \bar{I}_1 \\ \bar{I}_2\end{bmatrix} 
 = \begin{bmatrix} Y_l & -\dfrac{Y_l}{t} \\ -\dfrac{Y_l}{t^\star} & \dfrac{Y_l}{|t|^2} \end{bmatrix}
 \begin{bmatrix} \bar{V}_1 \\ \bar{V}_2\end{bmatrix}$$
\end{frame}

\begin{frame}[allowframebreaks]{System $\mathbf{Y}_{\text{bus}}$ Stamping \& Newton-Raphson Impact}
For a transformer connected between \alert{Bus $i$} (series impedance side) and \alert{Bus $j$} (ideal tap side $1:t$):

$$\begin{bmatrix} \bar{I}_i \\ \bar{I}_j \end{bmatrix} = 
\begin{bmatrix} 
Y_l & -\dfrac{Y_l}{t} \\ 
-\dfrac{Y_l}{t^\star} & \dfrac{Y_l}{|t|^2} 
\end{bmatrix}
\begin{bmatrix} \bar{V}_i \\ \bar{V}_j \end{bmatrix}$$

\textbf{Stamping Rule into the Global $\mathbf{Y}_{\text{bus}}$ Matrix:}
\begin{align*}
    Y_{ii} &\leftarrow Y_{ii} + Y_l \\
    Y_{jj} &\leftarrow Y_{jj} + \frac{Y_l}{|t|^2} \\
    Y_{ij} &\leftarrow Y_{ij} - \frac{Y_l}{t} \\
    Y_{ji} &\leftarrow Y_{ji} - \frac{Y_l}{t^\star}
\end{align*}

\newpage
\begin{alertblock}{Consequence for Complex Taps ($t = m e^{j\theta}, \theta \ne 0$)}
Because $t \ne t^\star$, the off-diagonal terms differ:
$$Y_{ij} \ne Y_{ji} \implies \mathbf{Y}_{\text{bus}} \ne \mathbf{Y}_{\text{bus}}^T \quad \text{(\alert{loss of symmetry!})}$$
\end{alertblock}

\textbf{Direct Impact on the Newton-Raphson Power Flow:}
\begin{itemize}
    \item Standard textbook formulations often assume $Y_{km} = Y_{mk}$ ($G_{km} = G_{mk}, B_{km} = B_{mk}$).
    \item With PSTs, one \alert{must} strictly use directed branch admittance values:
    $$P_k = \sum_{m=1}^n V_k V_m \left( G_{km} \cos(\theta_k - \theta_m) + B_{km} \sin(\theta_k - \theta_m) \right)$$
    where $G_{km} + j B_{km} = Y_{km} \ne Y_{mk}$.
    \item Off-diagonal blocks in the Jacobian $\mathbf{J}$ are no longer transpose-symmetric.
\end{itemize}
\end{frame}

\begin{frame}{The Equivalent $\pi$-Model for In-Phase Taps ($t \in \mathbb{R}$)}
When $t \in \mathbb{R}$ (pure voltage magnitude regulation, no phase shift), $t = t^\star$, so $Y_{ij} = Y_{ji}$. The two-port is reciprocal and can be synthesized as a \alert{passive $\pi$-equivalent}:

\begin{columns}[T]
    \begin{column}{0.56\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.75, >=Stealth]
            % Bus bars
            \draw[line width=2.5pt] (0, -1.0) -- (0, 1.1) node[above, font=\bfseries] {Bus 1 ($\bar{V}_1$)};
            \draw[line width=2.5pt] (5.5, -1.0) -- (5.5, 1.1) node[above, font=\bfseries] {Bus 2 ($\bar{V}_2$)};
            
            % Series branch
            \draw[line width=1pt] (0, 0.6) -- (1.5, 0.6);
            \draw[fill=white, line width=1pt] (1.5, 0.25) rectangle (4.0, 0.95) node[midway] {$y_s = \dfrac{Y_l}{t}$};
            \draw[line width=1pt] (4.0, 0.6) -- (5.5, 0.6);
            
            % Shunt 1
            \draw[line width=1pt] (0.9, 0.6) -- (0.9, 0.1);
            \draw[fill=white, line width=1pt] (0.1, -0.5) rectangle (1.7, 0.1) node[midway, font=\footnotesize] {$y_{sh1} = Y_l\frac{1-t}{t}$};
            \draw[line width=1pt] (0.9, -0.5) -- (0.9, -0.85);
            \draw[line width=1pt] (0.6, -0.85) -- (1.2, -0.85);
            \draw[line width=1pt] (0.7, -0.95) -- (1.1, -0.95);
            
            % Shunt 2
            \draw[line width=1pt] (4.6, 0.6) -- (4.6, 0.1);
            \draw[fill=white, line width=1pt] (3.8, -0.5) rectangle (5.4, 0.1) node[midway, font=\footnotesize] {$y_{sh2} = Y_l\frac{t-1}{t^2}$};
            \draw[line width=1pt] (4.6, -0.5) -- (4.6, -0.85);
            \draw[line width=1pt] (4.3, -0.85) -- (4.9, -0.85);
            \draw[line width=1pt] (4.4, -0.95) -- (4.8, -0.95);
        \end{tikzpicture}
        \end{center}
    \end{column}
    \begin{column}{0.44\textwidth}
        \small
        \textbf{Equivalence with Matrix:}
        \begin{align*}
            Y_{11} &= y_s + y_{sh1} = Y_l \\
            Y_{22} &= y_s + y_{sh2} = \frac{Y_l}{t^2} \\
            Y_{12} &= Y_{21} = -y_s = -\frac{Y_l}{t}
        \end{align*}
        \alert{Advantage:} Standard transmission line $\pi$-routines can be directly reused!
        \vspace{0.1cm}
        
        \footnotesize
        \textbf{Note on PST ($t \in \mathbb{C}$):} Non-reciprocal ($Y_{12} \ne Y_{21}$), cannot be modeled by passive two-terminal elements.
    \end{column}
\end{columns}
\end{frame}

\textbf{Network Reduction (Kron / Norton Equivalent):}
Since no power is injected at auxiliary Bus 4 ($I_4 = 0$):
$$\mathbf{Y}_{\text{reduced}} = \mathbf{Y}_{aa} - \mathbf{Y}_{ab} Y_{44}^{-1} \mathbf{Y}_{ba}$$
This yields an exact $3 \times 3$ equivalent network for custom Newton-Raphson solvers.

\begin{block}{Key Takeaways}
\begin{itemize}
    \item Standard line routines can represent OLTCs via modified $\pi$-shunts.
    \item Phase shifters require explicit directed admittance stamping in $\mathbf{Y}_{\text{bus}}$ and Newton-Raphson Jacobians.
\end{itemize}
\end{block}
