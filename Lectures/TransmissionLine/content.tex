% Definition
\begin{frame}{Definition}
\begin{columns}[c]
    \begin{column}{0.66\textwidth}
    \begin{itemize}
        \item An (overhead) transmission \textbf{line} is a set of 3 bundles of conductors corresponding to the three phases of the system.
        \item Commonly used \textbf{voltages} range from 70 kV to 380 kV in Belgium (more where distances are larger).
        \item Minimum \textbf{distances} between conductors depend on the voltage level, and thus electrical properties also depend on the voltage level.
        \item Underground \textbf{cables} are more and more used. They can be modeled in a similar way as overhead transmission lines, but they have different properties.
    \end{itemize}

\begin{frame}{Skin effect: current density distribution}
\begin{columns}[c]
    \begin{column}{0.48\textwidth}
    \small
    Current density decays with depth $d$ from surface:
    \vspace{-0.4em}
    \[
        J(d) = J_0 \, e^{-d / \delta}, \quad \delta = \sqrt{\frac{\rho}{\pi f \mu}}

    \vspace{-0.6em}

        \item At depth $d = \delta$: $J(\delta) = J_0/e \approx 36.8\%$.

        \item Effective AC resistance $R_{\text{AC}} > R_{\text{DC}}$.
        \item \textbf{ACSR}: steel core in center carries negligible current, providing mechanical strength without losses!
    \end{itemize}

\section{Distributed model}
% Distributed parameter representation
\begin{frame}[allowframebreaks]{Distributed parameter representation}
    We consider that we are in sinusoidal steady state. 
    On a per-phase basis, the line can be represented as many chunks connected to each other:  
    \begin{center}
        \input{images/TL_distributed.tex}
    \end{center}
    \textbf{How do voltage and current evolve as a function of the position on the line?}

    \alert{Let's fix the origin of $x$ at the receiving end where $\bar{V}_R$ and $\bar{V}_R$ are assumed known.}

    \begin{itemize}
        \item As $R$ is small, $R$ is considered as \textit{lumped} (a discrete resistive element on one side of the line)
        \item $\frac{d\bar{V}(x)}{dx} = j\omega L \bar{I}(x)$
        \item $\frac{d\bar{I}(x)}{dx} = j\omega C \bar{V}(x)$
    \end{itemize}

    Hence $$\frac{d^2\bar{V}(x)}{dx^2} + \beta^2 \bar{V}(x) = 0$$
    with $\beta = \omega \sqrt{LC}$ the \textit{propagation constant}
\end{frame}
\begin{frame}[allowframebreaks]{Solution of the ODE}

    The previous equation has a solution of the type 
    $$\bar{V}(x) = \bar{V}_1 e^{\beta j x} + \bar{V}_2 e^{-\beta j x}.$$
    
        
    By derivation, the current is  $$\bar{I}(x) = (\bar{V}_1 e^{\beta j x} - \bar{V}_2 e^{-\beta j x}) / Z_c.$$
    
    With the \textbf{surge impedance} $$Z_c = \sqrt{\frac{L}{C}}.$$
    
    The boundary conditions at $x=0$, $$\bar{V}(0) = \bar{V}_R = V_R \angle 0,$$  and $$\bar{I}(0) = \bar{I}_R$$
    allow to determine constants $\bar{V}_1$ and $\bar{V}_2$, and finally
    $$
    \bar{V}(x) = \bar{V}_R \cos(\beta x) + j Z_c \bar{I}_R \sin(\beta x).
    $$
\end{frame}

\section{Surge impedance loading}
\begin{frame}{Closing the line on the surge impedance $Z_c$}
    If the line is assumed lossless and \textit{we close it with $Z_c$}, assuming $\bar{V}_R = V_R \angle 0$:
    \begin{center}
        \includegraphics[width=\textwidth]{images/SIL.png}
    \end{center}
    then the voltage \textit{magnitude is constant} over the line: $\bar{V}(x) = V_R e^{j\beta x}$, and only the \textit{angle increases with $x$}.
    Similar conclusion for $\bar{I}(x)$.
    \begin{itemize}
        \item Why? The reactive power consumed by the line is the same as the reactive power produced, everywhere.
    \end{itemize}
\end{frame}

% Zc and SIL
\begin{frame}{Surge impedance loading}
    $Z_c$ depends on the line characteristics/geometry and is, hence, mainly a function of the voltage level (distances between conductors, etc.).

    The surge impedance loading (SIL) is the power drawn by the load $Z_c$, which depends on the voltage level $V_{LL}$
    $$
    SIL = \frac{V^2_{LL}}{Z_c}
    $$


    Example: for $500 \text{ kV}$, $SIL \approx 1020 \text{ MW}.$ (Exercise: show it) 
\end{frame}

% Line loadability
\begin{frame}{Line loadability}
    The SIL gives an idea of the loadability of a line depending on its length:
    \begin{itemize}
        \item short line, $ l < 100 \text{ km}$
        \begin{itemize}
            \item load limit $= 3 \times SIL$
            \item thermal limit (See Section~\ref{sec:LR} on Line rating)
        \end{itemize}
        \item Medium length line, $ 100 \text{ km} < l < 300 \text{ km}$
        \begin{itemize}
            \item load limit $= 1.5$ to $3 \times SIL$
            \item voltage drop $< 5\%$
        \end{itemize}
        \item Long line, $ l > 300 \text{ km}$
        \begin{itemize}
            \item load limit $\approx 1 \times SIL$
            \item for system stability, the angle difference between line ends should stay $< 40^\circ$, see lecture on Transient stability.
        \end{itemize}
    \end{itemize}
\end{frame}


\section{Lumped transmission line model}
\begin{frame}{The $\pi$ model}
If $l$ is relatively small ($< 300 \text{ km}$), we can \textit{approximate} the line with \textbf{lumped} parameters,
\begin{columns}
    \begin{column}{0.55\textwidth}
    \begin{center}
        \includegraphics[width=\textwidth]{images/pi_1.png}
    \end{center}
    \end{column}
    \begin{column}{0.45\textwidth}
    with, by manipulation of the previous equations and assuming $\beta l$ small,
    \begin{itemize}
        \item $Z_{\text{series}} = R l +  j \omega L l$
        \item $\frac{Y_{\text{shunt}}}{2} = j \frac{\omega C l}{2}$
    \end{itemize}
    
    Remember that $R$, $L$ and $C$ are per km values.

    This $\pi$ model is symmetrical by design.
    \end{column}
\end{columns}

    
\end{frame}

% Frame 1: Thermal limits & ground clearance
% -----------------------------------------------------------------------------
\begin{frame}[allowframebreaks]{Thermal Limits \& Ground Clearance}

Transmission lines have a \textbf{maximum conductor temperature} $T_{\text{max}}$ (typically $75\text{--}90^\circ\text{C}$):

\begin{columns}[c]
    \begin{column}{0.4\textwidth}
    \footnotesize
    \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{Thermal elongation:} Current generates Joule losses ($R I^2$). The conductor expands ($\Delta L = \alpha L \Delta T$), causing sag to increase: $s \approx \frac{w L^2}{8 H}$.
        \item \textbf{Ground clearance violation:} Sag reduces distance to ground/trees $\implies$ risk of catastrophic \textbf{flashover} (major contributor to 2003 US blackout!).
        \item \textbf{Material annealing:} Sustained excessive temperature permanently degrades aluminum tensile strength.
    \end{itemize}
    \end{column}

        \input{images/sag_clearance.tex}
    \end{column}
\end{columns}

% Frame 2: Conductor Heat Balance
% -----------------------------------------------------------------------------
\begin{frame}{Conductor Heat Balance (IEEE 738 / CIGRE 601)}
\begin{columns}[c]
    \begin{column}{0.50\textwidth}
    \footnotesize
    In steady-state, conductor temperature $T_c$ satisfies:
    \vspace{-0.4em}
    \[
        \underbrace{q_J + q_s}_{\text{Heat gain}} = \underbrace{q_c + q_r}_{\text{Heat loss}}
    \]
    \vspace{-0.7em}
    \begin{itemize}\setlength{\itemsep}{0pt}
        \item \textbf{Joule heating:} $q_J = R(T_c) \cdot I^2$
        \item \textbf{Solar gain:} $q_s = \alpha_s \cdot S \cdot D$
        \item \textbf{Convective cooling:} $q_c = h_c(v_w, \phi) \cdot (T_c - T_a)$ \\
              (\alert{\textbf{wind speed $v_w$ is the dominant cooling factor!}})
        \item \textbf{Radiative cooling:} $q_r = \pi D \epsilon \sigma_B (T_c^4 - T_a^4)$
    \end{itemize}
    \vspace{0.2em}
    \textbf{Thermal ampacity limit:}
    \vspace{-0.5em}
    \[
        I_{\text{max}} = \sqrt{\frac{q_c(T_{\text{max}}, T_a, v_w, \phi) + q_r(T_{\text{max}}, T_a) - q_s}{R(T_{\text{max}})}}
    \]
    \end{column}
    \begin{column}{0.50\textwidth}
        \centering
        \input{images/heat_balance.tex}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Static (SLR) and Seasonal (SeLR) Line Rating}
\begin{columns}[c]
    \begin{column}{0.48\textwidth}
    \footnotesize
    \textbf{Static Line Rating (SLR):}
    \begin{itemize}\setlength{\itemsep}{1.5pt}
        \item Conservative worst-case assumption: high $T_a$ ($35\text{--}40^\circ$C), low wind ($v_w \approx 0.5\,$m/s), full sun.
        \item Constant safety baseline across the year.
    \end{itemize}
    \vspace{0.2em}
    \textbf{Seasonal Line Rating (SeLR):}
    \begin{itemize}\setlength{\itemsep}{1.5pt}
        \item Separate Summer/Winter ratings ($+40\%$ base ampacity in winter due to cold ambient air).
    \end{itemize}
    \vspace{0.2em}
    \textbf{The Opportunity for DLR:}
    \begin{itemize}\setlength{\itemsep}{1.5pt}
        \item Atmospheric cooling is much stronger $>95\%$ of time.
        \item A gentle \alert{\textbf{2 m/s breeze adds +43\% capacity}}, but static limits keep this trapped!
    \end{itemize}
    \end{column}

\begin{frame}{Dynamic Line Rating (DLR) \& Wind Power Synergy}
\begin{columns}[c]
    \begin{column}{0.48\textwidth}
    \footnotesize
    \textbf{Dynamic Line Rating (DLR / RTTR):}
    \begin{itemize}\setlength{\itemsep}{1.5pt}
        \item Continuously updates ampacity $I_{\text{DLR}}(t)$ from real-time and forecasted weather/line state.
        \item \textbf{Wind Synergy:} Strong wind generation coincides with high convective cooling: \alert{\textbf{capacity unlocks when needed most!}}
    \end{itemize}
    \vspace{0.2em}
    \textbf{Operational Value:}
    \begin{itemize}\setlength{\itemsep}{1.5pt}
        \item \textbf{Headroom:} Unlocks $+10\text{--}50\%$ capacity on existing lines without new construction.
        \item \textbf{Safety:} Detects rare calm heatwaves where static ratings overestimate ampacity ($I_{\text{DLR}} < I_{\text{SLR}}$).
    \end{itemize}
    \end{column}
